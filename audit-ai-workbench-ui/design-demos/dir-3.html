-- 用途：识别采购付款中的拆分、超比例与节点异常WITH payment_base AS (  SELECT    p.payment_id, p.contract_id,    p.vendor_id, p.payment_date,    p.payment_amount, c.contract_amount,    SUM(p.payment_amount) OVER (      PARTITION BY p.contract_id      ORDER BY p.payment_date    ) AS cumulative_paid  FROM dwd_pay_order p  JOIN dwd_contract c    ON p.contract_id = c.contract_id  WHERE p.payment_date BETWEEN :start_date    AND :end_date), exception_flag AS (  SELECT *,    CASE WHEN cumulative_paid > contract_amount * 1.02      THEN '超合同金额'      WHEN payment_amount < :split_threshold      THEN '疑似拆分付款'    END AS risk_type  FROM payment_base)SELECT * FROM exception_flagWHERE risk_type IS NOT NULL;
