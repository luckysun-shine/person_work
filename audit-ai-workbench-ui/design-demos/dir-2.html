AI 逻辑摘要
先按合同汇总付款，再识别同供应商同日拆分以及累计付款超合同额 5% 的记录。该摘要由 AI 辅助生成，正式口径以 SQL 与人工说明为准。
purchase_payment_anomaly.sql
-- v3.2 · 采购付款异常明细（展示样例，不执行）
WITH payment_base AS (
  SELECT p.contract_id, p.supplier_id, p.pay_date,
         SUM(p.pay_amount) OVER (
           PARTITION BY p.contract_id
           ORDER BY p.pay_date
         ) AS accumulated_amount,
         c.contract_amount
  FROM fact_payment p
  JOIN dim_contract c ON p.contract_id = c.contract_id
  WHERE p.pay_date BETWEEN @start_date AND @end_date
)
SELECT contract_id, supplier_id, pay_date,
       accumulated_amount, contract_amount,
       accumulated_amount / NULLIF(contract_amount, 0) AS pay_ratio
FROM payment_base
WHERE accumulated_amount > contract_amount * 1.05;
